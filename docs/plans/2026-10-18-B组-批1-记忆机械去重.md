- [ ] **Step 3: 最小实现**

```python
#: 比较"是不是同一条记忆"时剥掉的首尾标点（全角 + 半角）。
_TRAILING_PUNCTUATION = "，。、；：！？,.!?;:"


def normalized_text(text: str) -> str:
    """归一化后比较"是不是同一条"：去全部空白 + 剥首尾标点。

    **只剥首尾**——中间的问号/逗号是内容的一部分（`assembly._asked_how` 剥标点同一口径）。
    只做**完全重复**，不做相似度（`02 §11.0.16`：相似度不做，要动得单独拍板）。
    """

    squeezed = "".join(text.split())
    return squeezed.strip(_TRAILING_PUNCTUATION)
```

### Task 2: `_reject` 去重判据 + 两个调用方传参

**Files:**

- Test: `backend/tests/domain/memory/test_memory_dedup.py`

**Interfaces:**
- Consumes: `normalized_text`（Task 1）
- Produces:
  `_reject(change, *, known_entries: set[str], known_refs: frozenset[str], taken: frozenset[tuple[str, str]]) -> tuple[str, DomainError] | None`
  —— `taken` 是已有条目的 `{(kind_value, normalized_text)}`。理由串固定为
  `"与已有记忆完全重复"`，错用 `ShapeError`。
- Produces: `screen_changes()` / `apply()` 内部各自算 `taken`；**批内**也进工作集。

- [ ] **Step 1: 写失败的测试**

```python
import pytest

from pmstudio.domain.memory.memory import (
    MemoryEntry, MemoryKind, MemoryOp, MemoryChange, ProjectMemory,
    screen_changes,
)
from pmstudio.domain.shared.errors import ShapeError


def _entry(memory_id, text, kind=MemoryKind.LESSON):
    from datetime import datetime
    from pmstudio.domain.shared.ids import EvidenceId
    return MemoryEntry(
        memory_id=memory_id, kind=kind, text=text,
        refs=(EvidenceId("E1"),), round_id="r1",
        created_at=datetime(2026, 10, 1), updated_at=datetime(2026, 10, 1),
    )


def _add(text, kind=MemoryKind.LESSON):
    return MemoryChange(op=MemoryOp.ADD, kind=kind, text=text, refs=("E1",))


class TestScreenChangesDedup:
    def test_exact_duplicate_is_dropped_with_reason(self):
        memory = ProjectMemory.empty("p1")
        memory = memory.__class__(project_id="p1", entries=(_entry("mem-1", "不要做多租户"),))
        result = screen_changes((_add("不要做多租户"),), memory=memory,
                                known_refs=frozenset({"E1"}))
        assert result.accepted == ()
        assert len(result.dropped) == 1
        assert "完全重复" in result.dropped[0].reason

    def test_normalized_duplicate_is_dropped(self):
        memory = ProjectMemory(project_id="p1", entries=(_entry("mem-1", "不要做多租户。"),))
        result = screen_changes((_add(" 不要做多租户 "),), memory=memory,
                                known_refs=frozenset({"E1"}))
        assert result.accepted == ()

    def test_same_text_different_kind_is_kept(self):
        memory = ProjectMemory(project_id="p1",
                               entries=(_entry("mem-1", "不要做多租户", MemoryKind.LESSON),))
        result = screen_changes((_add("不要做多租户", MemoryKind.REJECTED),), memory=memory,
                                known_refs=frozenset({"E1"}))
        assert len(result.accepted) == 1

    def test_near_duplicate_is_kept(self):
        """R：不引相似度——只在末尾多了三个字的**不是**重复。"""
        memory = ProjectMemory(project_id="p1", entries=(_entry("mem-1", "不要做多租户"),))
        result = screen_changes((_add("不要做多租户登录"),), memory=memory,
                                known_refs=frozenset({"E1"}))
        assert len(result.accepted) == 1

    def test_duplicate_inside_one_batch_is_dropped(self):
        """同一批里两条一模一样的，第二条也丢（工作集在批内长大）。"""
        memory = ProjectMemory.empty("p1")
        result = screen_changes((_add("不要做多租户"), _add("不要做多租户")),
                                memory=memory, known_refs=frozenset({"E1"}))
        assert len(result.accepted) == 1
        assert len(result.dropped) == 1

    def test_apply_rejects_duplicate_too(self):
        """写入侧用**同一条**判据（一处实现两处用）。"""
        memory = ProjectMemory(project_id="p1", entries=(_entry("mem-1", "不要做多租户"),))
        with pytest.raises(ShapeError):
            memory.apply((_add("不要做多租户"),), round_id="r1",
                         known_refs=frozenset({"E1"}),
                         now=__import__("datetime").datetime(2026, 10, 1), ids=None)
```

- [ ] **Step 3: 最小实现**

```python
def _taken_keys(entries: Sequence[MemoryEntry]) -> frozenset[tuple[str, str]]:
    """已有记忆的（类别, 归一化文本）集合——去重判据的输入，一处算。"""

    return frozenset(
        (str(entry.kind), normalized_text(entry.text)) for entry in entries
    )
```

`_reject` 增参并加判据（放在 `if change.op is not MemoryOp.DELETE:` 块**内**、`MAX_ENTRY_LINES` 之后）：

```python
        if change.op is MemoryOp.ADD and (
            str(change.kind), normalized_text(change.text)
        ) in taken:
            reason = "与已有记忆完全重复"
            return reason, ShapeError(reason)
```

`_reject` 签名末加 `taken: frozenset[tuple[str, str]] = frozenset()`。

`screen_changes` 里：`known = {...}` 那行后加

```python
    taken = set(_taken_keys(memory.entries))
```

并把循环里的 `_reject(...)` 调用补上 `taken=frozenset(taken)`；**接受一条 ADD 之后**加

```python
            if change.op is MemoryOp.ADD:
                taken.add((str(change.kind), normalized_text(change.text)))
```

`apply` 里同样先算 `taken = set(_taken_keys(self.entries))`，逐条传参、接受后更新工作集。
