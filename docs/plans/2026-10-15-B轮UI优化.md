```tsx
const [active, setActive] = useState("home-new");

    (entries) => {

      }
    },
    { rootMargin: "-40% 0px -55% 0px" },
  );

  return () => observer.disconnect();
}, []);
```

```ts
// progress.ts，statusText 旁边：
const LOOP_TASKS: readonly string[] = ["opening_card", "questionnaire", "draft_card"];

/** 右栏要不要展开：有待办的轮次才占 500px（06 §2.4 §3）。 */
export function needsLoopAttention(round: RoundView | null): boolean {
  return round !== null && LOOP_TASKS.includes(round.waiting_on);
}

/** rail 徽标数：未答题数 + 草案待确认组数（0/1）。 */
export function loopBadgeCount(round: RoundView | null): number {
  if (round === null) return 0;
  const done = new Set<string>();
  for (const batch of round.answer_sets ?? []) {
    for (const id of [...(batch.answered ?? []), ...(batch.skipped ?? [])]) done.add(id);
  }
  const open = (round.questions ?? []).filter((q) => !done.has(q.question_id)).length;
  return open + (round.draft !== null && round.draft !== undefined ? 1 : 0);
}
```

`answer_sets` 的 answered/skipped 字段名实现时按 `types.ts` 核对（`AnswerBatchView` 已有 `answerBatchSummary` 用例可抄）——字段名以类型为准，不猜。

```ts
// status.ts，sectionStatus 旁边：
export interface SectionBadge { block: number; warn: number }

/** 缺口戳计数：severity 原样分桶（DESIGN.md 状态色一套，计数不动语义）。 */
export function sectionBadge(
  sectionId: string,
  readiness: readonly { section_id: string; severity: string }[],
): SectionBadge | null {
  let block = 0;
  let warn = 0;
  for (const item of readiness) {
    if (item.section_id !== sectionId) continue;
    if (item.severity === "blocker") block += 1;
    else warn += 1;
  }
  return block === 0 && warn === 0 ? null : { block, warn };
}
```

`severity` 的取值（`"blocker"` vs 其他）实现时按 `types.ts` `ReadinessItemView` 核对——本计划里 Step 1 的 `"blocker"` 来自导出截图（`[blocker]`/`[warn]`），以类型为准。
